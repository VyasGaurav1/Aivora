cmake_minimum_required(VERSION 3.22.1)
project(aivora_reasoning LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_FLAGS_DEBUG "-g -O2" CACHE STRING "" FORCE)
set(CMAKE_CXX_FLAGS_DEBUG "-g -O2" CACHE STRING "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_OBJECT_PATH_MAX 180)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_BUILD_UI LLAMA_CONNECT LLAMA_OPENSSL GGML_NATIVE GGML_OPENMP GGML_OPENMP_FETCH GGML_RPC GGML_CPU_KLEIDIAI GGML_CPU_REPACK GGML_CPU_ALL_VARIANTS GGML_BACKEND_DL GGML_AVX GGML_AVX2 GGML_FMA GGML_F16C GGML_SSE42)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(LLAMA_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../../third_party/llama.cpp llama-build)
add_library(aivora_reasoning SHARED reasoning_jni.cpp)
target_compile_options(aivora_reasoning PRIVATE -fvisibility=hidden)
target_link_options(aivora_reasoning PRIVATE -Wl,--exclude-libs,ALL)
target_link_libraries(aivora_reasoning PRIVATE llama)
